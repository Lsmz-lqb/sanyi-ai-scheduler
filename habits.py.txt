"""本地习惯统计，只存脱敏标签，默认不上传任何数据"""
import json
from pathlib import Path

HABIT_FILE = Path("./local_habits.json")

class HabitTracker:
    def __init__(self):
        if not HABIT_FILE.exists():
            self.data = {"tags": {}, "heavy_count": 0, "light_count": 0}
            self.save()
        else:
            with open(HABIT_FILE, "r", encoding="utf-8") as f:
                self.data = json.load(f)

    def record_tag(self, tag: str, level: str):
        self.data["tags"][tag] = self.data["tags"].get(tag, 0) + 1
        if level == "heavy":
            self.data["heavy_count"] += 1
        else:
            self.data["light_count"] += 1
        self.save()

    def save(self):
        with open(HABIT_FILE, "w", encoding="utf-8") as f:
            json.dump(self.data, f, ensure_ascii=False, indent=2)

    def clear(self):
        self.data = {"tags": {}, "heavy_count": 0, "light_count": 0}
        self.save()
